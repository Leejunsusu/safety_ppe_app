# --- MySQL 데이터베이스 ---
# 본인이 MySQL 설치할 때 정한 root 비밀번호
spring.datasource.password=여기에_본인_MySQL_비밀번호

# --- JWT 토큰 비밀키 ---
# 토큰 서명에 쓰는 키. 반드시 32자 이상의 아무 문자열.
# 예시를 그대로 써도 개발엔 지장 없지만, 각자 다른 값을 권장.
jwt.secret=this-is-a-very-long-secret-key-change-me-36chars

# --- 이메일(Gmail SMTP) 발송 설정 ---
# host, port는 Gmail 공통이므로 아래 값 그대로 사용
spring.mail.host=smtp.gmail.com
spring.mail.port=587
# 본인 Gmail 주소
spring.mail.username=본인이메일@gmail.com
# Gmail '앱 비밀번호' 16자리 (일반 로그인 비밀번호 아님!)
# 발급법: 구글계정 → 보안 → 2단계 인증 켜기 → '앱 비밀번호' 검색 → 생성
spring.mail.password=구글_앱_비밀번호_16자리

# --- 이메일 인증 방식 (아래 두 줄은 그대로) ---
spring.mail.properties.mail.smtp.auth=true
spring.mail.properties.mail.smtp.starttls.enable=true
